async def test_dry_run_default_mode(test_staff_account):
    """Test that dry-run is the default mode and outputs correct JSON."""
    account = test_staff_account
    
    result = subprocess.run(
        [sys.executable, str(CLI_TOOL),
         '--account-id', account['id'],
         '--expected-generation', str(account['generation'])],
        capture_output=True,
        text=True
    )
    
    assert result.returncode == 0, f"CLI failed: {result.stderr}"
    
    # Parse output JSON
    output = json.loads(result.stdout.strip())
    
    # Verify dry-run output structure
    assert output['mode'] == 'DRY_RUN_NO_MUTATION'
    assert output['account_id'] == account['id']
    assert output['expected_generation'] == account['generation']
    assert output['next_generation'] == account['generation'] + 1
    assert 'warning' in output
    assert 'approved protected terminal' in output['warning']
    assert 'revokes prior staff sessions' in output['warning']
    
    # Verify no mutation occurred
    user = await users.find_one({'id': account['id']})
    assert user['credential_generation'] == account['generation']
    
    print(f"✓ Dry-run default mode outputs correct JSON without mutation")

async def test_multiple_accounts_different_generations():
    """Test dry-run with accounts at different generations."""
    # Create account at generation 0
    account1_id = str(uuid.uuid4())
    hashed = await asyncio.get_event_loop().run_in_executor(None, hashpw, 'Password123!')
    
    await users.insert_one({
        'id': account1_id,
        'email': f'multi-test-1-{uuid.uuid4().hex[:8]}@isolated.invalid',
        'name': 'Multi Test 1',
        'role': 'reviewer',
        'exam': 'JEE Main',
        'access_type': 'staff_password',
        'password_hash': hashed,
        'credential_generation': 0,
        'created_at': datetime.now(timezone.utc),
    })
    
    # Create account at generation 2
    account2_id = str(uuid.uuid4())
    await users.insert_one({
        'id': account2_id,
        'email': f'multi-test-2-{uuid.uuid4().hex[:8]}@isolated.invalid',
        'name': 'Multi Test 2',
        'role': 'owner',
        'exam': 'JEE Main',
        'access_type': 'staff_password',
        'password_hash': hashed,
        'credential_generation': 2,
        'created_at': datetime.now(timezone.utc),
    })
    
    try:
        # Test account 1 at generation 0
        result1 = subprocess.run(
            [sys.executable, str(CLI_TOOL),
             '--account-id', account1_id,
             '--expected-generation', '0'],
            capture_output=True,
            text=True
        )
        assert result1.returncode == 0
        output1 = json.loads(result1.stdout.strip())
        assert output1['expected_generation'] == 0
        assert output1['next_generation'] == 1
        
        # Test account 2 at generation 2
        result2 = subprocess.run(
            [sys.executable, str(CLI_TOOL),
             '--account-id', account2_id,
             '--expected-generation', '2'],
            capture_output=True,
            text=True
        )
        assert result2.returncode == 0
        output2 = json.loads(result2.stdout.strip())
        assert output2['expected_generation'] == 2
        assert output2['next_generation'] == 3
        
        print(f"✓ Multiple accounts at different generations handled correctly")
    finally:
        await users.delete_many({'id': {'$in': [account1_id, account2_id]}})
